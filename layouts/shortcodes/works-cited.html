{{/* Site-wide bibliography: renders every Zotero item that is cited anywhere on the site with {{< cite title="..." >}} or {{< popup-citation title="..." >}}. Cited titles are collected by scanning the raw content of every page at build time. Hugo renders pages in parallel and in no guaranteed order, so a shared store filled in by the cite shortcodes (site.Store) is not reliable here: this page may render before the pages it needs to know about. */}} {{- $context := . }} {{- $authorParam := $context.Get "authorNum" | default "3" }} {{- $authorNum := int $authorParam }} {{- /* Collect cited titles from every page's source. */ -}} {{- $citedTitles := slice }} {{- $shortcodeRE := `\{\{[<%]\s*(?:cite|popup-citation)\s[^>%]*?title\s*=\s*"([^"]+)"` }} {{- range site.Pages }} {{- range findRESubmatch $shortcodeRE .RawContent }} {{- $citedTitles = $citedTitles | append (index . 1) }} {{- end }} {{- end }} {{- $citedTitles = $citedTitles | uniq }} {{- /* Only named params other than authorNum act as filters (e.g. collections="X"). */ -}} {{- $singleCollection := false }} {{- if $context.IsNamedParams }} {{- if $context.Get "collections" }} {{- $singleCollection = true }} {{- end }} {{- end }} {{ $items := partial "zotero/fetch-items.html" (dict "groupId" site.Params.groupId) }} {{ $filtered := partial "zotero/filter-items.html" (dict "items" $items "params" $context.Params "onlyCited" true "citedTitles" $citedTitles ) }} {{ $citations := partial "zotero/build-citations.html" (dict "items" $filtered "authorNum" $authorNum ) }} {{ $collections := partial "zotero/build-collection-tree.html" site.Params.groupId }}
    {{- /* Cited items that are not in any collection */ -}} {{- range $citation := $citations }} {{- $cols := $citation.citation.collections | default (slice) }} {{- if eq (len $cols) 0 }} {{- partial "render/render-citation.html" . }} {{- end }} {{- end }} {{- /* Cited items grouped by collection (empty collections are skipped) */ -}} {{- range $collections }} {{- partial "render/render-collection.html" (dict "collection" . "citations" $citations "singleCollection" $singleCollection ) }} {{- end }}
